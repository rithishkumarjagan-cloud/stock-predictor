# DATA LOADING FUNCTION

@st.cache_data(ttl=3600)
def load_stock_data(symbol, years):
    """Load stock data from Yahoo Finance"""
    end_date = datetime.now()
    start_date = end_date - timedelta(days=years*365)
    
    with st.spinner(f'📥 Loading {symbol} data from Yahoo Finance...'):
        try:
            df = yf.download(symbol, start=start_date, end=end_date, progress=False)
            if df.empty:
                st.error(f"No data found for {symbol}")
                return None
            df = df[['Open', 'High', 'Low', 'Close', 'Volume']]
            df.dropna(inplace=True)
            return df
        except Exception as e:
            st.error(f"Error loading data: {e}")
            return None

# ============================================
# FEATURE ENGINEERING
# ============================================
def create_features(df):
    """Create technical indicators and features"""
    df_features = df.copy()
    
    # Moving Averages
    df_features['MA5'] = df_features['Close'].rolling(window=5).mean()
    df_features['MA10'] = df_features['Close'].rolling(window=10).mean()
    df_features['MA20'] = df_features['Close'].rolling(window=20).mean()
    df_features['MA50'] = df_features['Close'].rolling(window=50).mean()
    
    # Price features
    df_features['High_Low_Ratio'] = df_features['High'] / df_features['Low']
    df_features['Close_Open_Ratio'] = df_features['Close'] / df_features['Open']
    
    # Daily Returns
    df_features['Return'] = df_features['Close'].pct_change() * 100
    
    # Lag features
    df_features['Prev_Close'] = df_features['Close'].shift(1)
    df_features['Prev_Volume'] = df_features['Volume'].shift(1)
    df_features['Prev_Return'] = df_features['Return'].shift(1)
    
    # Rolling statistics
    df_features['Rolling_Mean_5'] = df_features['Close'].rolling(window=5).mean()
    df_features['Rolling_Std_5'] = df_features['Close'].rolling(window=5).std()
    df_features['Volume_MA5'] = df_features['Volume'].rolling(window=5).mean()
    
    # Volatility
    df_features['Volatility'] = df_features['Return'].rolling(window=20).std()
    
    # Drop NaN values
    df_features.dropna(inplace=True)
    
    return df_features

    rf_pred = rf_model.predict(X_test_scaled)
    rf_mae = mean_absolute_error(y_test, rf_pred)

    lr_pred = lr_model.predict(X_test_scaled)
    lr_mae = mean_absolute_error(y_test, lr_pred)
